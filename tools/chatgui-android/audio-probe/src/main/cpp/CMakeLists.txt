cmake_minimum_required(VERSION 3.22)
project(call_audio_probe LANGUAGES C CXX)
get_filename_component(UTUN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
add_library(call_audio_probe SHARED probe.cpp "${UTUN_ROOT}/lib/debug_config.c")
target_include_directories(call_audio_probe PRIVATE "${UTUN_ROOT}/lib")
target_compile_features(call_audio_probe PRIVATE cxx_std_17)
target_compile_options(call_audio_probe PRIVATE -Wall -Wextra -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_options(call_audio_probe PRIVATE -Wl,--gc-sections)
target_link_libraries(call_audio_probe aaudio log dl)
